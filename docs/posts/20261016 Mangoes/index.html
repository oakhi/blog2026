# Robust data parsing helper function
process_market_data <- function(display_name, symbol, category_group) {
  env_obj_name <- gsub("^\\^", "", symbol)
  
  if (!exists(env_obj_name)) {
    warning(paste("Symbol object not found in environment:", env_obj_name))
    return(NULL)
  }
  
  xts_obj <- get(env_obj_name)
  
  df <- data.frame(
    Date   = index(xts_obj),
    Close  = as.numeric(xts_obj[, 4]), # Closing price/level
    Volume = as.numeric(xts_obj[, 5])  # Trading volume
  ) %>%
    na.omit() %>%
    filter(Date >= START_DATE & Date <= END_DATE) %>%
    arrange(Date) %>%
    mutate(
      Index_Name       = display_name,
      Asset_Name       = display_name,  # Alias to ensure backwards compatibility
      Category         = category_group,
      Normalized_Close = (Close / first(Close)) * 100, # Base 100 normalization
      Daily_Return_Pct = (Close - lag(Close)) / lag(Close) * 100,
      Volume_M         = Volume / 1e6,
      Direction        = ifelse(Daily_Return_Pct >= 0, "Up", "Down")
    )
  
  return(df)
}

# Process both asset categories
df_indices_tmp <- bind_rows(lapply(names(INDEX_TICKERS), function(n) {
  process_market_data(n, INDEX_TICKERS[[n]], "Global Market Index")
}))

df_mangos_tmp <- bind_rows(lapply(names(MANGOS_TICKERS), function(n) {
  process_market_data(n, MANGOS_TICKERS[[n]], "MANGO-S AI Cluster")
}))

# Master Combined Dataframe
